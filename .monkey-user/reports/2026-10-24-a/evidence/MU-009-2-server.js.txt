  141 | // Serve um arquivo de public/ se ele existir. Devolve false para o chamador
  142 | // cair no SPA fallback (rotas do React Router, por exemplo).
  143 | async function tryServePublicFile(res, pathname) {
  144 |   let decoded;
  145 |   try {
  146 |     decoded = decodeURIComponent(pathname);
  147 |   } catch {
  148 |     return false;
  149 |   }
  150 |   if (decoded.includes('\0')) return false;
  151 | 
  152 |   // `.` + pathname mantém o caminho relativo; o startsWith impede que
  153 |   // `..`/`%2e%2e` escapem da raiz pública e leiam .env ou o banco.
  154 |   const filePath = resolve(PUBLIC_DIR, `.${decoded}`);
  155 |   if (filePath !== PUBLIC_DIR && !filePath.startsWith(PUBLIC_DIR + sep)) return false;
  156 | 
  157 |   try {
  158 |     const info = await stat(filePath);
  159 |     if (!info.isFile()) return false;
  160 |   } catch {
  161 |     return false;
  162 |   }
  163 | 
  164 |   const headers = {};
  165 |   if (decoded === '/sw.js' || decoded === '/manifest.webmanifest' || decoded === '/index.html') {
  166 |     // O service worker e o manifest precisam ser revalidados a cada deploy.
  167 |     headers['Cache-Control'] = 'no-cache';
  168 |   } else if (decoded.startsWith('/assets/')) {
  169 |     // Nomes com hash: cache de 1 ano é seguro.
  170 |     headers['Cache-Control'] = 'public, max-age=31536000, immutable';
  171 |   }
  172 | 
  173 |   await sendFile(res, filePath, headers);
  174 |   return true;
  175 | }
