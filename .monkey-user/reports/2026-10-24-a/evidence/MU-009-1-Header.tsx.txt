   47 | function formatCountdown(nextRun: number | null): string {
   48 |   if (!nextRun) return '--:--';
   49 |   const diff = nextRun - Date.now();
   50 |   if (diff <= 0) return 'now';
   51 |   const mins = Math.floor(diff / 60000);
   52 |   const secs = Math.floor((diff % 60000) / 1000);
   53 |   return `${mins}m ${secs}s`;
   54 | }
