   12 |   /** Offline snapshot: read state and bookmarks cannot be written. */

   20 | /**
   21 |  * Leitor interno: mostra o texto que já veio no RSS, sem buscar a página
   22 |  * original. Nada aqui dispara requisição externa — nem imagens — então o
   23 |  * leitor funciona offline e não abre abas.
   24 |  *
   25 |  * O HTML do feed nunca é injetado: `cleanDescription()` remove as tags e o
   26 |  * resultado é renderizado como texto.
   27 |  */
   28 | export function ReaderModal({
   29 |   item,
   30 |   isBookmarked,
   31 |   readOnly = false,
   32 |   onClose,
   33 |   onToggleBookmark,
   34 |   onReadToggle,
   35 |   onShare,
   36 | }: ReaderModalProps) {
   37 |   const { t, language } = useI18n();
   38 |   const dialogRef = useRef<HTMLDivElement>(null);
   39 |   useEscapeKey(true, onClose);
   40 | 
   41 |   // Focus trap: o leitor é uma camada modal, então o Tab não pode escapar
   42 |   // para o header/cards atrás dele. Ao fechar, devolve o foco ao card.
   43 |   useEffect(() => {
   44 |     const previous = document.activeElement as HTMLElement | null;
   45 |     const dialog = dialogRef.current;
   46 |     if (!dialog) return;
   47 | 
   48 |     const focusable = () =>
   49 |       Array.from(
   50 |         dialog.querySelectorAll<HTMLElement>(
   51 |           'a[href], button:not([disabled]), input:not([disabled]), select:not([disabled]), textarea:not([disabled]), [tabindex]:not([tabindex="-1"])'
   52 |         )
   53 |       ).filter((element) => element.offsetParent !== null);
   54 | 
   55 |     dialog.focus();
   56 | 
   57 |     const onKeyDown = (event: KeyboardEvent) => {
   58 |       if (event.key !== 'Tab') return;
   59 | 
   60 |       const elements = focusable();
   61 |       if (elements.length === 0) {
   62 |         event.preventDefault();
   63 |         return;
   64 |       }
   65 | 
   66 |       const first = elements[0];
   67 |       const last = elements[elements.length - 1];
   68 |       const active = document.activeElement;
   69 | 
   70 |       if (event.shiftKey && (active === first || active === dialog)) {
   71 |         event.preventDefault();
   72 |         last.focus();
   73 |       } else if (!event.shiftKey && active === last) {
   74 |         event.preventDefault();
   75 |         first.focus();
   76 |       }
   77 |     };
   78 | 
   79 |     dialog.addEventListener('keydown', onKeyDown);
   80 |     return () => {
   81 |       dialog.removeEventListener('keydown', onKeyDown);
   82 |       previous?.focus?.();
   83 |     };
   84 |   }, []);
   85 | 
   86 |   const locale = language === 'en' ? 'en-US' : 'pt-BR';
   87 |   const isRead = Boolean(item.readAt);
   88 |   const body = cleanDescription(item.description || '') || cleanDescription(item.content || '');
   89 | 
   90 |   const formatDate = (iso: string) => {
   91 |     try {
   92 |       return new Date(iso).toLocaleString(locale, {
   93 |         day: '2-digit',
   94 |         month: 'long',
   95 |         year: 'numeric',
   96 |         hour: '2-digit',
   97 |         minute: '2-digit',
   98 |       });
   99 |     } catch {
  100 |       return iso;
  101 |     }
  102 |   };
  103 | 
  104 |   return (
  105 |     <div className="modal-overlay reader-overlay" onClick={onClose}>
  106 |       <div
  107 |         ref={dialogRef}
  108 |         className="modal reader-modal"
  109 |         role="dialog"
  110 |         aria-modal="true"
  111 |         aria-labelledby="reader-title"
  112 |         tabIndex={-1}
  113 |         onClick={(event) => event.stopPropagation()}
  114 |       >
  115 |         <div className="modal-header reader-header">
  116 |           <div className="reader-byline">
  117 |             <span className="news-source">{item.source || t('news.unknownSource')}</span>
  118 |             <time dateTime={item.pubDate}>{formatDate(item.pubDate)}</time>
  119 |           </div>
  120 |           <button className="close-btn" onClick={onClose} aria-label={t('common.close')}>
  121 |             <X size={18} />
  122 |           </button>
  123 |         </div>
  124 | 
  125 |         <div className="modal-body reader-body">
  126 |           <h2 id="reader-title" className="reader-title">
  127 |             {item.title}
  128 |           </h2>
  129 | 
  130 |           <div className="reader-text">
  131 |             {body || <span className="muted">{t('news.noDescription')}</span>}
  132 |           </div>
  133 | 
  134 |           <p className="reader-truncated">
  135 |             {t('reader.truncated')}{' '}
  136 |             <a href={item.link} target="_blank" rel="noopener noreferrer">
  137 |               {t('reader.readInSource')}
  138 |             </a>
  139 |           </p>
  140 |         </div>
  141 | 
  142 |         <div className="modal-footer reader-actions">
  143 |           <button
  144 |             className="btn btn-read"
  145 |             onClick={() => onReadToggle(item, !isRead)}
  146 |             disabled={readOnly}
  147 |             aria-label={t(isRead ? 'news.markUnreadAria' : 'news.markReadAria', {
  148 |               title: item.title,
  149 |             })}
  150 |             title={t(isRead ? 'news.markUnread' : 'news.markRead')}
  151 |           >
  152 |             {isRead ? <MailOpen size={17} /> : <Mail size={17} />}
  153 |             <span className="btn-label">{t(isRead ? 'news.markUnread' : 'news.markRead')}</span>
  154 |           </button>
  155 | 
  156 |           <BookmarkButton
  157 |             newsId={item.id}
  158 |             isBookmarked={isBookmarked}
  159 |             onToggle={onToggleBookmark}
  160 |             disabled={readOnly}
  161 |           />
  162 | 
  163 |           <button className="btn btn-share" onClick={() => onShare(item)}>
  164 |             <Share2 size={15} />
  165 |             <span className="btn-label">{t('news.share')}</span>
  166 |           </button>
  167 | 
  168 |           <a
  169 |             href={item.link}
  170 |             target="_blank"
  171 |             rel="noopener noreferrer"
  172 |             className="btn btn-primary reader-original"
  173 |           >
  174 |             <ExternalLink size={15} />
  175 |             <span className="btn-label">{t('reader.openOriginal')}</span>
  176 |           </a>
  177 |         </div>
  178 |       </div>
  179 |     </div>
  180 |   );
  181 | }
